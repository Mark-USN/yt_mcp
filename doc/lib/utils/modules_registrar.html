 45class RegistrarBase:
 46    """ Base class for different types of registrars (tools, prompts, resources). """
 47    def __init__(self, mcp: FastMCP, app_ctx: RuntimeContext, module_dir: Path | str) -> None:
 48        """ Initialize the registrar with the MCP server, application context, and module directory.
 49            Args:
 50                mcp (FastMCP): The MCP server instance to register tools/prompts/resources with.
 51                app_ctx (RuntimeContext): The runtime context of the application.
 52                module_dir (Path | str): The directory containing the modules to be registered.
 53        """
 54        self.mcp_server = mcp
 55        self.ctx = app_ctx
 56        self.base_dir = app_ctx.app_dir.parents[1].resolve()
 57        self.module_dir = Path(module_dir).resolve()
 58
 59        if not self.module_dir.is_dir():
 60            raise ModuleLoaderError(f"Not a directory: {self.module_dir}")
 61
 62    def register(self) -> None:
 63        """ Abstract method to be implemented by subclasses to perform the actual registration 
 64            logic. 
 65        """
 66        raise NotImplementedError

 80    def register_modules_with_server(self, module: ModuleType) -> None:
 81        """ Register all registered methods from a specific module.
 82            Args:
 83                module (ModuleType): The module containing a register(mcp) method.
 84        """
 85        if not hasattr(module, "register"):
 86            logger.warning("Module %s has no register(mcp) function", module.__name__)
 87            return
 88
 89        module.register(self.mcp_server, self.ctx)
 90        logger.info("🔧 Registered methods from %s", module.__name__)
 91
 92    def discover_modules(self, package: str) -> List[ModuleType]:
 93        """ Discover all module libs inside the given package.
 94            Args:
 95                package (str): Python package path containing the python libs.
 96            Returns:
 97                List[ModuleType]: A list of successfully imported lib.
 98            Raises:
 99                ModuleDiscoveryError: If the package cannot be imported or no modules are found.
100        """
101        try:
102            pkg = importlib.import_module(package)
103        except ImportError as exc:
104            logger.error("Could not import package '%s': %s", package, exc)
105            return []
106
107        modules: List[ModuleType] = []
108
109        for _, modname, ispkg in pkgutil.iter_modules(pkg.__path__):
110            # TODO: MCP does not handle submodules. Need to add code to recurse
111            # into subpackages and 'flatten' them into the main package namespace.
112            if ispkg :
113                continue
114
115            full_name = f"{package}.{modname}"
116            try:
117                module = importlib.import_module(full_name)
118                modules.append(module)
119                logger.info("Loaded method lib: %s into module %s", full_name, package)
120            except Exception as exc:      # pylint: disable=broad-exception-caught
121                logger.exception("Error importing lib %s: %s", full_name, exc)
122                continue
123
124        return modules
125
126
127    def package_name_from_module_dir(self) -> str:
128        """ Derive the Python package name from the module directory path, relative to the
129            base directory.
130            Returns:
131                str: The Python package name.
132            Raises:
133                ModuleLoaderError: If the module directory is not under the base directory.
134        """
135        root = self.base_dir.resolve()
136        module_dir = self.module_dir.resolve()
137
138        try:
139            relative_dir = module_dir.relative_to(root)
140        except ValueError as exc:
141            raise ModuleLoaderError(
142                f"module_dir must be under base_dir:\n"
143                f"  module_dir: {module_dir}\n"
144                f"  base_dir:   {root}"
145            ) from exc
146
147        return ".".join(relative_dir.parts)
148
149
150    def load_package_from_directory(self) -> tuple[ModuleType, str]:
151        """ Load the Python package from the module directory, ensuring it has an __init__.py
152            file.
153            Returns:
154                tuple[ModuleType, str]: The imported package module and its name.
155        """
156        package_name = self.package_name_from_module_dir()
157
158        init_file = self.module_dir / "__init__.py"
159        if not init_file.is_file():
160            raise ModuleLoaderError(f"Missing package file: {init_file}")
161
162        base_text = str(self.base_dir)
163        if base_text not in sys.path:
164            sys.path.insert(0, base_text)
165
166        package = importlib.import_module(package_name)
167        return package, package_name
168
169    def register(self) -> None:
170        """ Register all discovered and qualifying .py files with the MCP server. """
171        _, module_name = self.load_package_from_directory()
172
173        modules = self.discover_modules(module_name)
174        if not modules:
175            logger.warning("No method lib found in package '%s'", module_name)
176
177        for module in modules:
178            self.register_modules_with_server(module)

191    def normalize_params(self, raw_params: Any) -> dict[str, dict[str, Any]]:
192        """ Normalize the 'params' block from YAML into a dict:
193            { name: {description, required, default, type} }
194            Args:
195                raw_params: The raw 'params' block from YAML.
196            Returns:
197                A dictionary of normalized parameters.
198            Accepts either:
199              - a mapping: params: { text: { ... }, lang: { ... } }
200              - or a list of dicts: params: [ {name: text, ...}, {name: lang, ...} ]
201        """
202        if not raw_params:
203            return {}
204
205        params: dict[str, dict[str, Any]] = {}
206
207        # Case 1: mapping style
208        if isinstance(raw_params, dict):
209            for name, cfg in raw_params.items():
210                if cfg is None:
211                    cfg = {}
212                elif not isinstance(cfg, dict):
213                    # e.g. params: {text: "some description"}
214                    cfg = {"description": str(cfg)}
215
216                name_str = str(name)
217                params[name_str] = {
218                    "description": cfg.get("description"),
219                    "required": bool(cfg.get("required", True)),
220                    "default": cfg.get("default"),
221                    "type": cfg.get("type", "string"),
222                }
223            return params
224
225        # Case 2: list-of-dicts style
226        if isinstance(raw_params, list):
227            for item in raw_params:
228                if not isinstance(item, dict):
229                    continue
230                name = item.get("name")
231                if not name:
232                    continue
233                name_str = str(name)
234                params[name_str] = {
235                    "description": item.get("description"),
236                    "required": bool(item.get("required", True)),
237                    "default": item.get("default"),
238                    "type": item.get("type", "string"),
239                }
240            return params
241
242        logger.warning("Unsupported 'params' format in front matter: %r", raw_params)
243        return {}

245    def make_dynamic_prompt_fn(self, name: str, prompt_body: str, params: dict[str, dict]):
246        """ Create a function with a real dynamic signature that FastMCP accepts.
247            Args:
248                name: The name of the function to create.
249                prompt_body: The body of the prompt template with {placeholders}.
250                params: A dict of parameter metadata, keyed by parameter name.
251            Returns:
252                A callable function that renders the prompt with parameters.
253        """
254
255        # Build the function signature text
256        parts = []
257        for p, cfg in params.items():
258            if "default" in cfg:
259                default = repr(cfg["default"])
260                parts.append(f"{p}={default}")
261            else:
262                parts.append(p)
263        arglist = ", ".join(parts) if parts else ""
264
265        # Build the function body code
266        code = f'''
267def {name}({arglist}):
268    values = locals()
269    try:
270        return """{prompt_body}""".format_map(values)
271    except KeyError as exc:
272        raise ValueError(
273            f"Missing value for template placeholder: {{exc}}"
274        )
275'''
276
277        # Prepare namespace
278        ns = {}
279
280        # Execute the function definition
281        exec(textwrap.dedent(code), ns)     # pylint: disable=exec-used
282
283        # Extract the function object
284        return ns[name]

286    def register_markdown_prompt(self, md_path: Path) -> None:
287        """ Register a single markdown prompt file with the MCP server.
288            Args:
289                md_path: Path to the markdown file containing the prompt definition.
290        """
291        try:
292            post = frontmatter.load(md_path)  # parses YAML front matter if present
293        except Exception as exc:      # pylint: disable=broad-exception-caught
294            logger.exception("Failed to parse front matter in %s: %s", md_path, exc)
295            return
296
297        body: str = str(post.content).strip()
298        meta: dict[str, Any] = dict(post.metadata or {})
299
300        # Core fields
301        name: str = meta.get("name") or md_path.stem
302        style: str = meta.get("style", "plain")
303        description: str = meta.get("description") or f"Render '{name}' prompt ({style})."
304
305        # Tags: allow string or list, always ensure "public"
306        raw_tags = meta.get("tags") or ["public"]
307        if isinstance(raw_tags, str):
308            tags = {raw_tags}
309        else:
310            tags = {str(t).strip() for t in raw_tags if str(t).strip()}
311        if "public" not in tags:
312            tags.add("public")
313
314        # Params: normalize to a stable dict format
315        raw_params = meta.get("params")
316        params_meta = self.normalize_params(raw_params)
317
318        # Extra meta: everything not core
319        extra_meta = {
320            k: v
321            for k, v in meta.items()
322            if k not in {"name", "description", "tags", "style", "params"}
323        }
324
325        fn = self.make_dynamic_prompt_fn(name, body, params_meta)
326
327        # Register the prompt with FastMCP
328        self.mcp_server.prompt(
329            name=name,
330            description=description,
331            tags=tags,
332            meta={
333                "style": style,
334                "source_file": md_path.name,
335                "params": params_meta,  # expose param metadata to clients
336                **extra_meta,
337            },
338        )(fn)
339
340        logger.info("Registered prompt '%s' from %s", name, md_path.name)
341
342    def register_markdown_prompts(self) -> None:
343        """ Discover and register all markdown prompt files in the module directory."""
344        for md_path in self.module_dir.rglob("*.md"):
345            self.register_markdown_prompt(md_path)
346
347    def register(self) -> None:
348        """ Register all discovered and qualifying .py files with the MCP server and
349            parse and register markdown prompts.
350        """
351        PyRegistrar(self.mcp_server, self.ctx, self.module_dir).register()
352        self.register_markdown_prompts()

360class ResourcesRegistrar(RegistrarBase):
361    """ Register designated methods in any py files in the target directory and then 
362        register static resource files (e.g. .txt, .json) in the given directory as MCP resources. 
363    """
364    TEXT_SUFFIXES = {
365        ".md": "text/markdown",
366        ".txt": "text/plain",
367        ".json": "application/json",
368        ".xml": "application/xml",
369        ".html": "text/html",
370        ".css": "text/css",
371        ".js": "text/javascript",
372    }
373
374
375    def register_file_resource(self, path: Path, mime_type: str) -> None:
376        """ Register a single file as a FileResource with the MCP server.
377            Args:
378                path (Path): The path to the file to be registered.
379                mime_type (str): The MIME type of the file.
380        """
381        rel = path.relative_to(self.module_dir).as_posix()
382        uri = f"file://{rel}"
383        file_resource = FileResource(
384                path=path,
385                uri=uri,
386                name=path.stem,
387                description=f"Static resource from {rel}",
388                mime_type=mime_type,
389                tags={"public", "resource", "static"},
390            )
391        result = self.mcp_server.add_resource(file_resource)
392        logger.info("Registered file resource '%s' at URI %s Result %s", path.name, uri, result)
393
394    def register_text_resource(self, path: Path) -> None:
395        """ Register a single text file as a TextResource with the MCP server.
396            Args:
397                path (Path): The path to the file to be registered.
398                mime_type (str): The MIME type of the file.
399        """
400        rel = path.relative_to(self.module_dir).as_posix()
401        uri = f"resource://{rel}"
402        with path.open("r", encoding="utf-8-sig") as f:
403            content = f.read()
404        text_resource = TextResource(
405            uri=uri,
406            name=path.stem,
407            text=content,
408            tags={"public", "notification"}
409        )
410        result = self.mcp_server.add_resource(text_resource)
411        logger.info("Registered text resource '%s' at URI %s Result %s", path.name, uri, result)
412
413    def register_directory_resource(self, path: Path) -> None:
414        """ Register a directory with the MCP server.
415            Args:
416                path (Path): The path to the directory to be registered.
417        """
418        # data_dir_path = path.relative_to(self.module_dir.parent).resolve()
419        uri = f"resource://{path.stem}"
420        if path.is_dir():
421            data_listing_resource = DirectoryResource(
422                uri=uri,
423                path=path, # Path to the directory
424                name=f"{path.stem} Directory Listing",
425                description=f"Lists files available in the {path.stem} directory.",
426                recursive=False # Set to True to list subdirectories
427            )
428            result = self.mcp_server.add_resource(data_listing_resource)
429            logger.info("Registered directory resource '%s' at URI %s Result %s",
430                        path.name, uri, result)
431
432    def register(self) -> None:
433        """ Register all .py files and static text resources with the MCP server. """
434        PyRegistrar(
435            self.mcp_server,
436            self.ctx,
437            self.module_dir,
438        ).register()
439
440        for suffix, mime_type in self.TEXT_SUFFIXES.items():
441            for path in self.module_dir.rglob(f"*{suffix}"):
442                if "text" in mime_type:
443                    self.register_text_resource(path)
444                else:
445                    self.register_file_resource(path, mime_type)
446        self.register_directory_resource(self.module_dir / "Public")

192    def normalize_params(self, raw_params: Any) -> dict[str, dict[str, Any]]:
193        """ Normalize the 'params' block from YAML into a dict:
194            { name: {description, required, default, type} }
195            Args:
196                raw_params: The raw 'params' block from YAML.
197            Returns:
198                A dictionary of normalized parameters.
199            Accepts either:
200              - a mapping: params: { text: { ... }, lang: { ... } }
201              - or a list of dicts: params: [ {name: text, ...}, {name: lang, ...} ]
202        """
203        if not raw_params:
204            return {}
205
206        params: dict[str, dict[str, Any]] = {}
207
208        # Case 1: mapping style
209        if isinstance(raw_params, dict):
210            for name, cfg in raw_params.items():
211                if cfg is None:
212                    cfg = {}
213                elif not isinstance(cfg, dict):
214                    # e.g. params: {text: "some description"}
215                    cfg = {"description": str(cfg)}
216
217                name_str = str(name)
218                params[name_str] = {
219                    "description": cfg.get("description"),
220                    "required": bool(cfg.get("required", True)),
221                    "default": cfg.get("default"),
222                    "type": cfg.get("type", "string"),
223                }
224            return params
225
226        # Case 2: list-of-dicts style
227        if isinstance(raw_params, list):
228            for item in raw_params:
229                if not isinstance(item, dict):
230                    continue
231                name = item.get("name")
232                if not name:
233                    continue
234                name_str = str(name)
235                params[name_str] = {

237                    "required": bool(item.get("required", True)),
238                    "default": item.get("default"),
239                    "type": item.get("type", "string"),
240                }
241            return params

243        logger.warning("Unsupported 'params' format in front matter: %r", raw_params)
244        return {}

274            f"Missing value for template placeholder: {{exc}}"
275        )
276'''
277
278        # Prepare namespace
279        ns = {}
280

282        exec(textwrap.dedent(code), ns)     # pylint: disable=exec-used
283
284        # Extract the function object
285        return ns[name]
